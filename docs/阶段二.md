## 🎯 **前提约定（必须双方确认）**

### **1. 数据库使用约定**
```
数据库：ry-vue（现有数据库，A已配置好）
规则：
- A负责的表：term, course, course_class, credit_rule
- B负责的表：enrollment, class_time_slot
- 互不修改对方表结构
- 插入数据加前缀标识（A: a_, B: b_）
```

### **2. 数据前缀规则**
```sql
-- A插入的数据：
INSERT INTO term (term_id, term_name) VALUES ('a_2024-2025-1', 'A学期');

-- B插入的数据：  
INSERT INTO enrollment (student_id, class_id) VALUES (2001, 1);
-- enrollment不需要前缀，因为A不会操作此表

-- B查询时过滤自己的数据：
SELECT * FROM term WHERE term_id LIKE 'a_%';  -- A的数据
SELECT * FROM term WHERE term_id NOT LIKE 'a_%' AND term_id NOT LIKE 'b_%'; -- 公共数据
```

### **3. 备份约定**
```
每天开发前备份：
mysqldump -u root -p ry-vue > backup/ry-vue_日期.sql

重要操作前备份：
ALTER TABLE前、大量DELETE前必须备份
```

```
# 创建存储过程脚本文件
cat > sql/edu_procedures.sql << 'EOF'
-- 文件：edu_procedures.sql
-- 位置：D:/coding/project/school-teaching-system/sql/edu_procedures.sql
-- 描述：选课系统存储过程

USE ry-vue;

-- 删除已有的存储过程（如果存在）
DROP PROCEDURE IF EXISTS sp_calc_student_credits;
DROP FUNCTION IF EXISTS fn_get_student_credits;

DELIMITER $$

-- 存储过程：计算学生某学期已选课程总学分
-- 契约：GET /edu/statistics/student/{id}/credits
CREATE PROCEDURE sp_calc_student_credits(
    IN p_student_id BIGINT,
    IN p_term_id VARCHAR(20),
    OUT p_total_credits INT
)
BEGIN
    DECLARE v_total INT DEFAULT 0;
    
    -- 计算学生在该学期已选课程的总学分
    -- 注意：term_id可能需要处理前缀
    SELECT COALESCE(SUM(c.credit_hours), 0) INTO v_total
    FROM enrollment e
    JOIN course_class cc ON e.class_id = cc.class_id
    JOIN course c ON cc.course_id = c.course_id
    WHERE e.student_id = p_student_id
      AND e.term_id LIKE CONCAT('%', p_term_id, '%')  -- 模糊匹配，兼容前缀
      AND e.status = 'ENROLLED'
      AND cc.status = 1;  -- 只计算开放选课的课程
    
    SET p_total_credits = v_total;
END$$

-- 存储过程：检查学生是否可以选课（学分上限+时间冲突）
CREATE PROCEDURE sp_check_enrollment_eligible(
    IN p_student_id BIGINT,
    IN p_class_id INT,
    OUT p_is_eligible BOOLEAN,
    OUT p_error_message VARCHAR(200)
)
BEGIN
    DECLARE v_current_credits INT DEFAULT 0;
    DECLARE v_course_credits INT DEFAULT 0;
    DECLARE v_max_credits INT DEFAULT 20;
    DECLARE v_term_id VARCHAR(20);
    
    -- 获取课程信息和学期
    SELECT cc.term_id, c.credit_hours INTO v_term_id, v_course_credits
    FROM course_class cc
    JOIN course c ON cc.course_id = c.course_id
    WHERE cc.class_id = p_class_id;
    
    -- 计算当前已选学分
    CALL sp_calc_student_credits(p_student_id, v_term_id, v_current_credits);
    
    -- 获取学分上限
    SELECT COALESCE(MAX(max_credits), 20) INTO v_max_credits
    FROM credit_rule 
    WHERE term_id = v_term_id AND is_active = 1;
    
    -- 检查学分上限
    IF v_current_credits + v_course_credits > v_max_credits THEN
        SET p_is_eligible = FALSE;
        SET p_error_message = CONCAT('超过学分上限：当前', v_current_credits, '学分，本课程', v_course_credits, '学分，上限', v_max_credits, '学分');
    ELSE
        SET p_is_eligible = TRUE;
        SET p_error_message = NULL;
    END IF;
END$$

-- 函数：获取学生学分（方便SQL中调用）
CREATE FUNCTION fn_get_student_credits(p_student_id BIGINT, p_term_id VARCHAR(20))
RETURNS INT
DETERMINISTIC
READS SQL DATA
BEGIN
    DECLARE v_credits INT;
    CALL sp_calc_student_credits(p_student_id, p_term_id, v_credits);
    RETURN v_credits;
END$$

DELIMITER ;
```

```
// 文件：src/main/java/com/ruoyi/edu/admin/service/impl/CreditServiceImpl.java
package com.ruoyi.edu.admin.service.impl;

import com.ruoyi.edu.admin.mapper.CreditMapper;
import com.ruoyi.edu.admin.service.ICreditService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.HashMap;
import java.util.Map;

@Service
public class CreditServiceImpl implements ICreditService {
    
    @Autowired
    private CreditMapper creditMapper;
    
    @Override
    public Integer calculateStudentCredits(Long studentId, String termId) {
        try {
            // 方法1：调用存储过程
            Map<String, Object> params = new HashMap<>();
            params.put("p_student_id", studentId);
            params.put("p_term_id", termId);
            params.put("p_total_credits", null);
            
            creditMapper.callCalcStudentCredits(params);
            
            Integer credits = (Integer) params.get("p_total_credits");
            return credits != null ? credits : 0;
            
        } catch (Exception e) {
            // 方法2：降级到函数调用
            return creditMapper.getStudentCredits(studentId, termId);
        }
    }
    
    @Override
    public Map<String, Object> checkEnrollmentEligible(Long studentId, Long classId) {
        Map<String, Object> result = new HashMap<>();
        
        try {
            Map<String, Object> params = new HashMap<>();
            params.put("p_student_id", studentId);
            params.put("p_class_id", classId);
            params.put("p_is_eligible", null);
            params.put("p_error_message", null);
            
            creditMapper.callCheckEnrollmentEligible(params);
            
            result.put("isEligible", params.get("p_is_eligible"));
            result.put("errorMessage", params.get("p_error_message"));
            result.put("success", true);
            
        } catch (Exception e) {
            // 存储过程调用失败，返回默认值
            result.put("isEligible", true);  // 开发阶段默认允许选课
            result.put("errorMessage", "存储过程调用失败，使用默认检查");
            result.put("success", false);
        }
        
        return result;
    }
    
    @Override
    public Map<String, Object> getStudentCreditInfo(Long studentId, String termId) {
        Map<String, Object> result = new HashMap<>();
        
        // 计算当前学分
        Integer currentCredits = calculateStudentCredits(studentId, termId);
        
        // 获取学分上限（从credit_rule表，开发阶段用默认值）
        Integer maxCredits = 20;  // 默认值
        
        result.put("studentId", studentId);
        result.put("termId", termId);
        result.put("currentCredits", currentCredits);
        result.put("maxCredits", maxCredits);
        result.put("remainingCredits", Math.max(0, maxCredits - currentCredits));
        result.put("percentage", maxCredits > 0 ? (currentCredits * 100.0 / maxCredits) : 0);
        result.put("status", currentCredits >= maxCredits ? "已满" : "可选");
        
        return result;
    }
}
```

```sql
-- 存储过程：计算学生学分
-- 契约：GET /edu/statistics/student/{id}/credits
CREATE PROCEDURE sp_calc_student_credits(
    IN p_student_id BIGINT,
    IN p_term_id VARCHAR(20),
    OUT p_total_credits INT
)
BEGIN
    DECLARE v_total INT DEFAULT 0;
    
    -- 计算学生在该学期已选课程的总学分
    SELECT COALESCE(SUM(c.credit_hours), 0) INTO v_total
    FROM enrollment e
    JOIN course_class cc ON e.class_id = cc.class_id
    JOIN course c ON cc.course_id = c.course_id
    WHERE e.student_id = p_student_id
      AND e.term_id = p_term_id
      AND e.status = 'ENROLLED';
    
    SET p_total_credits = v_total;
END$$

DELIMITER ;

-- 创建调用存储过程的函数
CREATE FUNCTION fn_get_student_credits(p_student_id BIGINT, p_term_id VARCHAR(20))
RETURNS INT
DETERMINISTIC
READS SQL DATA
BEGIN
    DECLARE v_credits INT;
    CALL sp_calc_student_credits(p_student_id, p_term_id, v_credits);
    RETURN v_credits;
END;
```

### **接口契约（需要双方确认）**

#### **A需要提供的接口（告诉B）：**
```yaml
# 学期管理
GET    /edu/admin/terms                     # 获取所有学期

GET    /edu/admin/class/{id}/stats          # 班级统计（调用B的接口）
```

#### **B需要提供的接口（告诉A）：**
```yaml
# 学生功能
POST   /edu/student/enroll                  # 选课
POST   /edu/student/drop/{id}               # 退课
GET    /edu/student/timetable               # 个人课表

# 教师功能
GET    /edu/teacher/classes/{id}/students   # 班级学生名单
POST   /edu/teacher/grades/batch            # 批量录入成绩
POST   /edu/teacher/grades/publish          # 发布成绩

# 统计功能
GET    /edu/statistics/student/{id}/credits # 学分统计（调用存储过程）
```

### **数据前缀约定（需要确认）**
```sql
-- 双方确认数据前缀规则：
A的前缀：'a_'（所有A插入的数据加此前缀）
B的前缀：'b_'（B操作A的表时加此前缀）

-- 示例：
-- A插入学期：
INSERT INTO term (term_id, term_name) VALUES ('a_2024-2025-1', '2024-2025秋季');

-- B插入学期（如果需要）：
INSERT INTO term (term_id, term_name) VALUES ('b_test_term', 'B的测试学期');

-- 查询时：
-- A查询：WHERE term_id LIKE 'a_%'
-- B查询：WHERE term_id LIKE 'b_%' OR term_id NOT LIKE 'a_%' AND term_id NOT LIKE 'b_%'
```

```bash
#!/bin/bash
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/path/to/backup"
```
