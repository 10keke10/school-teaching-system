-- 1. 创建容量控制触发器（在Navicat中执行）
DELIMITER $$
CREATE TRIGGER trg_before_enroll_insert
BEFORE INSERT ON enrollment
FOR EACH ROW
BEGIN
    DECLARE v_capacity INT;
    DECLARE v_selected INT;
    DECLARE v_term_active BOOLEAN;
    
    -- 检查课程容量
    SELECT capacity, selected_count INTO v_capacity, v_selected
    FROM course_class WHERE class_id = NEW.class_id;
    
    IF v_selected >= v_capacity THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = '课程容量已满';
    END IF;
    
    -- 检查是否在选课时间内（简化：检查学期是否active）
    SELECT is_active INTO v_term_active
    FROM term WHERE term_id = NEW.term_id;
    
    IF NOT v_term_active THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = '不在选课时间内';
    END IF;
END$$
DELIMITER ;

-- 2. 创建选课人数更新触发器
DELIMITER $$
CREATE TRIGGER trg_after_enroll_change
AFTER INSERT ON enrollment
FOR EACH ROW
BEGIN
    IF NEW.status = 'ENROLLED' THEN
        UPDATE course_class 
        SET selected_count = selected_count + 1
        WHERE class_id = NEW.class_id;
    END IF;
END$$
DELIMITER ;

-- 3. 创建学分计算存储过程
DELIMITER $$
CREATE PROCEDURE sp_calc_student_credits(
    IN p_student_id INT,
    IN p_term_id VARCHAR(20),
    OUT p_total_credits INT
)
BEGIN
    SELECT COALESCE(SUM(c.credit_hours), 0) INTO p_total_credits
    FROM enrollment e
    JOIN course_class cc ON e.class_id = cc.class_id
    JOIN course c ON cc.course_id = c.course_id
    WHERE e.student_id = p_student_id
      AND e.term_id = p_term_id
      AND e.status = 'ENROLLED';
END$$
DELIMITER ;

// 文件：EnrollmentServiceImpl.java
@Service
public class EnrollmentServiceImpl implements IEnrollmentService {
    
    @Autowired
    private EnrollmentMapper enrollmentMapper;
    
    @Autowired
    private CourseClassMapper courseClassMapper;
    
    @Transactional(rollbackFor = Exception.class)
    public AjaxResult enroll(Long studentId, Long classId) {
        // 步骤1：检查是否已选
        if (enrollmentMapper.exists(studentId, classId)) {
            return AjaxResult.error("不能重复选课");
        }
        
        // 步骤2：获取课程信息
        CourseClass courseClass = courseClassMapper.selectById(classId);
        
        // 步骤3：检查学分上限（调用存储过程）
        Integer currentCredits = callCalcCreditsProcedure(studentId, courseClass.getTermId());
        Integer courseCredits = getCourseCredits(courseClass.getCourseId());
        
        if (currentCredits + courseCredits > getMaxCredits(courseClass.getTermId())) {
            return AjaxResult.error("超过学分上限");
        }
        
        // 步骤4：检查时间冲突
        if (hasTimeConflict(studentId, classId)) {
            return AjaxResult.error("时间冲突");
        }
        
        // 步骤5：插入记录（触发器会检查容量）
        Enrollment enrollment = new Enrollment();
        enrollment.setStudentId(studentId);
        enrollment.setClassId(classId);
        enrollment.setTermId(courseClass.getTermId());
        enrollment.setStatus("ENROLLED");
        
        int rows = enrollmentMapper.insert(enrollment);
        return rows > 0 ? AjaxResult.success() : AjaxResult.error();
    }
    
    // 时间冲突检测（简化版）
    private boolean hasTimeConflict(Long studentId, Long newClassId) {
        // 获取学生已选课程的时间
        List<String> occupiedTimes = enrollmentMapper.selectStudentSchedule(studentId);
        // 获取新课程的时间
        String newClassTime = courseClassMapper.selectTimeById(newClassId);
        
        // 简单的时间字符串匹配（可按需复杂化）
        for (String time : occupiedTimes) {
            if (timeConflict(time, newClassTime)) {
                return true;
            }
        }
        return false;
    }
}
